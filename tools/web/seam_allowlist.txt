# x86-only sources that seam_scan.py --forbid accepts. See docs/web_build.md section 4.9.
#
# Line format:  pattern [kind,kind,...]  # reason
#   pattern  path glob relative to the repository root ('*' within a directory, '**' across them);
#            a leading '!' removes the path from the effect of earlier lines.
#   kinds    the kinds the line covers (default: all). The groups guard, platform and os cover
#            the win32-call hits of that group.
# The last matching line wins.

tools/reference-game/invitro/startup_reservation.h  # the Windows claim of the guest range before the CRT

# Windows guards: the vectored trap for unconverted calls, the CPU fault translation, the
# host-fault handler and the image-access measurement stay on Windows.
sim/src/cpu_fault.S

# The guest memory accessor: the one place a guest VA becomes a host pointer, the proxy's
# unchecked identity accessor and es2sim's checked one (docs/web_build.md section 5).
tools/reference-game/proxy/guest_native.h   guest-host-cast  # the identity mapping of the accessor
sim/src/guest_memory.c                      guest-host-cast  # the identity mapping of the checked accessor
# A GMEM handle of the headless audio is a host heap pointer the guest holds unchanged.
sim/src/audio.c                             guest-host-cast  # free() of the handle calloc() gave
# The web build's claim of the guest range is the image base of linear memory itself.
sim/src/os_process_wasm.c                   guest-host-cast  # the image base as the reserved range
